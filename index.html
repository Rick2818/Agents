    async function loginWithGoogle() {
      if (supabaseClient && supabaseClient.auth) {
        try {
          const { error } = await supabaseClient.auth.signInWithOAuth({
            provider: 'google',
            options: { redirectTo: window.location.href }
          });
          if (error) throw error;
          return;
        } catch (err) {
          console.warn('OAuth Google fallback to direct login:', err.message);
        }
      }
      // Demo fallback login
      // Sesion OAuth
      localStorage.removeItem('antigravity_user');
      initUserProfile();
      switchLanguage(localStorage.getItem('destraba_lang') || 'es');

      toggleUserDropdown();
    }

    function updateFinancialRoiCard(painText, agentSpec) {
      const roiCard = document.getElementById('financial-roi-card');
      if (!roiCard) return;

      let lossUsd = 1480;
      let planRate = 79;
      let hours = 85;

      const lower = (painText || '').toLowerCase();
      if (lower.includes('whatsapp') || lower.includes('soporte') || lower.includes('cliente')) {
        lossUsd = 1250;
        planRate = 49;
        hours = 90;
      } else if (lower.includes('cobranza') || lower.includes('factura') || lower.includes('moros')) {
        lossUsd = 2100;
        planRate = 89;
        hours = 65;
      } else if (lower.includes('inventario') || lower.includes('stock') || lower.includes('entrega')) {
        lossUsd = 1680;
        planRate = 69;
        hours = 80;
      } else if (lower.includes('ventas') || lower.includes('prospectos') || lower.includes('lead')) {
        lossUsd = 2800;
        planRate = 79;
        hours = 110;
      }

      const netSavings = lossUsd - planRate;
      const roiMultiple = ((netSavings / planRate) * 10).toFixed(1);

      document.getElementById('roi-loss-amount').textContent = `$${lossUsd.toLocaleString()} USD/mes`;
      document.getElementById('roi-plan-price').textContent = `$${planRate} USD/mes`;
      document.getElementById('roi-net-savings').textContent = `$${netSavings.toLocaleString()} USD/mes`;
      document.getElementById('roi-explanation').innerHTML = `Tener a tu personal atendiendo este cuello de botella cuesta ~${hours} horas/mes de nómina improductiva. Activar este agente fiduciario lo erradica en 60 segundos con un Retorno de Inversión (ROI) estimado de ${roiMultiple}x.`;

      roiCard.classList.remove('hidden');
    }
